cmake_minimum_required(VERSION 3.16)
project(mc_option_pricer CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")
find_package(Threads REQUIRED)
include_directories(include)
add_executable(test_pricer tests/test_pricer.cpp)
target_link_libraries(test_pricer Threads::Threads)
add_executable(experiments src/experiments.cpp)
target_link_libraries(experiments Threads::Threads)
enable_testing()
add_test(NAME pricer COMMAND test_pricer)
